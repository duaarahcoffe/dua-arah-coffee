let cart = []; function add(name, price) { let item = cart.find(i => i.name === name); if (item) { item.qty += 1; } else { cart.push({ name, price, qty: 1, note: "" }); } render(); } function decrease(name) { let item = cart.find(i => i.name === name); if (!item) return; item.qty -= 1; if (item.qty <= 0) { cart = cart.filter(i => i.name !== name); } render(); } function updateNote(name, value) { let item = cart.find(i => i.name === name); if (item) { item.note = value; } } function render() { let html = ""; let total = 0; if (cart.length === 0) { document.getElementById("cart").innerHTML = "Kosong"; return; } cart.forEach(i => { total += i.price * i.qty; html += `
${i.name}
Rp ${i.price.toLocaleString('id-ID')} x ${i.qty}

`; }); html += `Total: Rp ${total.toLocaleString('id-ID')}`; document.getElementById("cart").innerHTML = html; } function toggleCart() { document.getElementById("sidebar").classList.toggle("active"); } function paymentCheck() { let val = document.getElementById("payment").value; if (val === "qris") { openModal("QRIS Payment", "qris.jpg"); } if (val === "bca") { openModal("Transfer BCA", "bca.jpg"); } } function openModal(title, img) { document.getElementById("title").innerText = title; document.getElementById("img").src = img; document.getElementById("modal").style.display = "flex"; } function closeModal() { document.getElementById("modal").style.display = "none"; } function checkout() { let nama = document.getElementById("nama").value; let alamat = document.getElementById("alamat").value; if (cart.length === 0) { alert("Keranjang kosong"); return; } if (!nama || !alamat) { alert("Isi data lengkap"); return; } let text = "Halo Dua Arah Coffee,%0A%0A"; cart.forEach(i => { text += `- ${i.name} x${i.qty}%0A`; if (i.note) { text += ` Catatan: ${i.note}%0A`; } }); text += `%0ANama: ${nama}%0AAlamat: ${alamat}`; window.open("https://wa.me/6288210129982?text=" + text); }
